% ---------------------------------------------------------------- stat: la domanda
\begin{frame}{\`E un vero legame o \`e un caso?}
  \vfill
  \begin{itemize}[<+->]
    \setlength\itemsep{1.0em}
    \item Vediamo error rate diversi tra gli orientamenti.
    \item Ma potrebbe essere solo \hl{rumore} dei dati.
    \item Serve un \hl{test statistico} che dica: questo legame \`e reale?
  \end{itemize}
  \vfill
  \onslide<4->{\begin{center}\small Strumento usato: il \hl{test del chi-quadro ($\chi^2$) di indipendenza}.\end{center}}
\end{frame}

% ---------------------------------------------------------------- stat: idea del chi2
\begin{frame}{Passo 2 --- l'idea del chi-quadro}
  \vfill
  \begin{itemize}[<+->]
    \setlength\itemsep{0.9em}
    \item Immaginiamo che l'orientamento \hl{non c'entri} con gli errori.
    \item Allora ogni gruppo dovrebbe sbagliare \hl{nella stessa proporzione} (circa la media, $\sim$42\%).
    \item Il $\chi^2$ misura \hl{quanto} gli errori \emph{osservati} si allontanano da questo scenario ``senza legame''.
    \item Pi\`u grande \`e lo scarto $\rightarrow$ $\chi^2$ grande $\rightarrow$ pi\`u probabile che un legame ci sia.
  \end{itemize}
  \vfill
\end{frame}

% ---------------------------------------------------------------- stat: p-value
\begin{frame}{Passo 3 --- il $p$-value}
  \vfill
  \begin{itemize}[<+->]
    \setlength\itemsep{0.9em}
    \item Un $\chi^2$ grande potrebbe capitare \hl{per caso}?
    \item Il \hl{$p$-value} \`e la probabilit\`a di vedere uno scarto cos\`i grande \emph{se non ci fosse nessun legame}.
    \item $p$ piccolissimo (es. $p<0.001$) $\rightarrow$ quasi impossibile che sia un caso.
    \item Quindi il legame \`e \hl{reale}, non fortuna.
  \end{itemize}
  \vfill
\end{frame}

% ---------------------------------------------------------------- stat: Cramer V
\begin{frame}{Passo 4 --- quanto \`e forte? (Cram\'er's V)}
  \vfill
  \begin{itemize}[<+->]
    \setlength\itemsep{0.9em}
    \item Il $p$-value dice \hl{se} c'\`e un legame, non \hl{quanto} \`e forte.
    \item Con tanti dati anche un legame \hl{piccolo} risulta ``significativo''.
    \item \hl{Cram\'er's V} misura la \hl{forza}: 0 = nessun legame, 1 = legame massimo.
  \end{itemize}
  \vfill
\end{frame}

\begin{frame}{Revisore 1 --- (1) Novit\`a metodologica limitata}
  \vfill
  \begin{center}
  \itshape\small
  ``The two-stage framework is essentially standard fine-tuning of pretrained language models followed by majority voting. The main contribution seems to be the task formulation, dataset construction, and evaluation setup, rather than a new model architecture, learning objective, or aggregation method.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 1 --- (2) Il guadagno dell'aggregazione non \`e misurato pulito}
  \vfill
  \begin{center}
  \itshape\small
  ``The comparison between the article-level score ($\sim$0.60) and the publisher-level score ($\sim$0.69) changes both the evaluation unit and the weighting of publishers. As a result, the improvement cannot be attributed to aggregation alone.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 1 --- (3) Bias confuso con il livello di affidabilit\`a}
  \vfill
  \begin{center}
  \itshape\small
  ``The distribution of reliability labels differs substantially across political leanings, and some reliability levels are clearly more difficult to classify than others. The relationship between political leaning and prediction error should therefore be tested again after controlling for the true reliability level.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 1 --- (4) Il chi-quadro ignora il clustering}
  \vfill
  \begin{center}
  \itshape\small
  ``Political leaning and reliability are publisher-level attributes shared by multiple articles from the same publisher. Using all articles as independent observations may lead to pseudo-replication and overstate the effective sample size and significance of the results.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 1 --- (5) Indizi dell'editore e artefatti non esclusi}
  \vfill
  \begin{center}
  \itshape\small
  ``The paper does not include systematic text cleaning, source-identifier masking, or sanitized-text ablations. It is therefore difficult to tell whether the model is learning reliability-related content or simply exploiting publisher-specific signals and scraping artifacts.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 1 --- (6) Possibile bias di selezione nello scraping}
  \vfill
  \begin{center}
  \itshape\small
  ``Only 439 of the 669 candidate publishers were included in the final dataset, and the scraping success rates differ across reliability levels. The paper does not examine whether the publishers that were successfully scraped differ systematically from those that were not.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 1 --- (7) Confronto sulla taglia del modello non controllato}
  \vfill
  \begin{center}
  \itshape\small
  ``The models differ in architecture, fine-tuning strategy, training schedule, and context length, in addition to parameter count. The observed performance differences therefore cannot be attributed to model size alone.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 1 --- (8) Nessuna stima di incertezza}
  \vfill
  \begin{center}
  \itshape\small
  ``Differences such as 0.69 versus 0.68 are small, but the paper does not report confidence intervals, bootstrap results, or paired significance tests. It is therefore unclear whether these differences are meaningful.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 2 --- (1) Motivazione debole del passaggio a livello editore}
  \vfill
  \begin{center}
  \itshape\small
  ``publisher reliability is, by construction, obtained as a fixed aggregation of article-level outputs\dots the re-formulation does not introduce substantial new insight beyond a change in the unit of evaluation.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 2 --- (2) Mancano confronti con altri scorer}
  \vfill
  \begin{center}
  \itshape\small
  ``Publisher-level scores can equally be produced by first scoring individual articles with a model pretrained specifically for article-level reliability classification, or with a general-purpose LLM such as Qwen, and then applying the same voting strategy\dots No such comparison is reported.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Revisore 2 --- (3) Il dataset non \`e ``large-scale''}
  \vfill
  \begin{center}
  \itshape\small
  ``Although the corpus contains 19,476 articles, the objective of the paper is scoring at the level of the publisher, and the dataset comprises only 439 publishers. The dataset is therefore unlikely to support effective use in real-world applications.''
  \end{center}
  \vfill
\end{frame}

\begin{frame}{Marinella $\times$ ChatGPT}
  \small
  \begin{itemize}[<+->]
    \setlength\itemsep{0.7em}
    \item \`E una revisione \hl{statistica} della sezione bias
    \item Il $\chi^2$ ad articolo non \`e un test valido come analisi principale (articoli non indipendenti) $\rightarrow$ test a livello di editore
    \item Leaning e reliability: problema di \hl{overlap/positivity}, non solo ``entangled'' --- ``controlling for reliability'' rischia di sembrare pi\`u conclusivo di quanto sia
    \item La MED calcolata solo sugli errori pu\`o fuorviare $\rightarrow$ riportare anche il bias su \emph{tutti} gli articoli
    \item Linguaggio: ``reference labels'' invece di ``ground truth''; cautela su ``small-to-moderate''
    \item Messaggio: stiamo misurando un'\hl{associazione}, non una discriminazione causata dal modello
  \end{itemize}
\end{frame}

\begin{frame}{Fabio $\times$ Claude --- (1) idee e review}
  \begin{itemize}[<+->]
    \setlength\itemsep{0.8em}
    \item Tre parti: idee sui modelli, una review completa, un'analisi quantitativa del bias
    \item \hl{Modelli}: gli LLM grandi ``conoscono'' le testate per nome (contaminazione) $\rightarrow$ i piccoli fine-tuned sono lo strumento giusto; baseline LLM con \hl{mascheramento dei nomi}
    \item Idee nuove: LLM come \hl{estrattore di feature} + classificatore; \hl{embeddings + multiple-instance learning}
    \item \hl{Review} (major revision): baseline deboli, aggregazione non studiata, incertezza mancante, bias confuso
  \end{itemize}
\end{frame}

\begin{frame}{Fabio $\times$ Claude --- (2) l'analisi del bias}
  \begin{itemize}[<+->]
    \setlength\itemsep{0.8em}
    \item Dalle matrici di confusione ha ricostruito l'errore per \hl{(livello $\times$ orientamento)}
    \item ``Slightly Left sottostimata'' \`e un \hl{artefatto di composizione} $\rightarrow$ va tolta
    \item ``Gli estremi sbagliano di pi\`u'' regge, ma perch\'e sono la \hl{minoranza} del loro livello
    \item Il modello ha imparato l'orientamento come \hl{scorciatoia} $\rightarrow$ spiega anche il nullo di Mistral
    \item \`E la riformulazione pi\`u forte della sezione bias
  \end{itemize}
\end{frame}

\begin{frame}{Cosa hanno in comune}
  \vfill
  \begin{itemize}[<+->]
    \setlength\itemsep{0.9em}
    \item Entrambe convergono col Revisore 1: il nodo \`e il \hl{confondimento} leaning/reliability e il \hl{clustering}
    \item Fabio va oltre: lo \hl{quantifica} e propone la riscrittura
    \item La sezione bias \`e il punto che decide il nuovo paper
  \end{itemize}
  \vfill
\end{frame}
